cmake_minimum_required(VERSION 3.16)
project(galaxy_native_bringup LANGUAGES CXX)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${RUNTIME_ARCHIVE}")
    message(FATAL_ERROR "A current verified Vita runtime archive is required")
endif()
add_executable(galaxy_bringup main.cpp renderer.cpp)
target_include_directories(galaxy_bringup PRIVATE "${root}/platform/include" "${root}/platform/src/gx")
target_compile_features(galaxy_bringup PRIVATE cxx_std_20)
target_compile_options(galaxy_bringup PRIVATE -O2 -mcpu=cortex-a9 -mfpu=neon -mfloat-abi=hard -mthumb
    -fsigned-char -fshort-wchar -fno-fast-math -ffunction-sections -fdata-sections)
set_source_files_properties(renderer.cpp PROPERTIES COMPILE_OPTIONS "-fno-short-enums")
set_source_files_properties(main.cpp PROPERTIES COMPILE_OPTIONS "-fshort-enums")
target_link_options(galaxy_bringup PRIVATE -Wl,--gc-sections -Wl,--no-undefined -Wl,-Map,galaxy_bringup.map)
target_link_libraries(galaxy_bringup PRIVATE "${RUNTIME_ARCHIVE}" SceGxm_stub SceDisplay_stub SceCtrl_stub
    SceMotion_stub SceTouch_stub SceKernelThreadMgr_stub SceSysmem_stub SceIofilemgr_stub SceProcessmgr_stub pthread m)
vita_create_self(eboot.bin galaxy_bringup)
vita_create_vpk(galaxy-native-bringup.vpk CDXG00001 eboot.bin VERSION 00.01 NAME "Galaxy Native Diagnostics")
